# Continuous-integration tooling, pinned for reproducible builds.
#
# These tools are used by the CI pipeline only. They are deliberately kept
# out of backend/requirements.txt so that a deployment never installs an
# auditing tool, a linter or a test runner it does not need, and so that a
# change of the auditing tool can never silently change the behaviour of a
# release build. The container image of the backend installs
# backend/requirements.txt alone and therefore carries none of them.
#
# Every version below is pinned to an exact release. The pipeline installs
# backend/requirements.lock first and this file second, so the pins in this
# file are the ones that end up installed, whatever the ranges of the runtime
# requirements resolve to. A tool upgrade is therefore an explicit, visible
# change to this file rather than a silent consequence of a new release of
# the package index.
#
# The pinned versions are the ones the pipeline was verified against: ruff
# 0.17.6, mypy 2.4.1 and pytest 9.2.0, on Python 3.12, 3.13 and 3.14.
#
# SECURITY: no credentials, no private keys and no connection details of real
# equipment belong in this file.

# Formatting and linting.
ruff==0.17.6

# Static type checking.
mypy==2.4.1
types-PyYAML==6.0.12.20261122

# Test execution and coverage measurement.
pytest==9.2.0
pytest-asyncio==1.5.0
pytest-cov==7.1.0

# The tool that measures the coverage the policy gate reads. It is pinned
# because pytest-cov only states a lower bound for it, and a coverage report
# produced by an unpinned version is a report the pipeline cannot reproduce:
# a tool upgrade would then change the numbers the gate decides on without
# any change to the repository.
coverage==7.17.5

# Property-based physics testing.
hypothesis==6.170.17

# Dependency vulnerability auditing.
pip-audit==2.10.1

# Version and requirement parsing. It is a transitive dependency of the test
# runner, but scripts/check_dependency_lock.py imports it directly to compare
# the pinned versions against the declared ranges, so it is declared here
# rather than relied upon transitively.
packaging==26.4
